-- Drop existing tables if any (to reset)
DROP TABLE IF EXISTS Students, Classes, Teachers, Subjects;

-- Create Subjects table
CREATE TABLE Subjects (
    subject_id INT PRIMARY KEY,
    subject_name VARCHAR(50)
);

-- Create Teachers table
CREATE TABLE Teachers (
    teacher_id INT PRIMARY KEY,
    name VARCHAR(100),
    subject VARCHAR(50)
);

-- Create Classes table
CREATE TABLE Classes (
    class_id INT PRIMARY KEY,
    class_name VARCHAR(50),
    teacher_id INT,
    FOREIGN KEY (teacher_id) REFERENCES Teachers(teacher_id)
);

-- Create Students table
CREATE TABLE Students (
    student_id INT PRIMARY KEY,
    name VARCHAR(100),
    age INT,
    class_id INT,
    FOREIGN KEY (class_id) REFERENCES Classes(class_id)
);

1,1️⃣ LEFT JOIN – classes with students

SELECT *
FROM classes
LEFT JOIN students
ON students.class_id = classes.class_id;


2.2️⃣ Subject–Teacher combinations (CROSS JOIN)

SELECT s.subject_name, t.teacher_name
FROM subjects s
CROSS JOIN teachers t;

3️⃣ Student–Subject combinations

SELECT st.student_name, sub.subject_name
FROM students st
CROSS JOIN subjects sub;

4️⃣ Students + class name + teacher name (if available)

SELECT st.student_name, c.class_name, t.teacher_name
FROM students st
LEFT JOIN classes c
ON st.class_id = c.class_id
LEFT JOIN teachers t
ON c.class_id = t.class_id;


5️⃣ Classes with teachers and students
SELECT c.class_name, t.teacher_name, st.student_name
FROM classes c
LEFT JOIN teachers t
ON c.class_id = t.class_id
LEFT JOIN students st
ON c.class_id = st.class_id;
